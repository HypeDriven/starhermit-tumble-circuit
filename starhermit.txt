# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.24.0
engine.minor=23
engine.capabilities=platform,storage,clipboard

# Keyboard actions (the browser build's names, so bindings saved with the
# account carry over; players rebind keys and pads in the runtime's Options).
# On touch screens the runtime draws a movement stick (bottom left, mirrored
# for left-handed touch controls) and Jump / Dive buttons during a round.
control.left=KeyA+ArrowLeft | Move left
control.right=KeyD+ArrowRight | Move right
control.forward=KeyW+ArrowUp | Move forward
control.back=KeyS+ArrowDown | Move back
control.jump=Space+KeyK | Jump
control.dive=ShiftLeft+ShiftRight+KeyL | Dive
control.pause=Escape+KeyP | Pause
engine.group.left=Racer
engine.group.pause=Round

# Gamepad defaults (the browser build's mapping): the left stick or D-pad
# runs, South jumps, East or West dives, Start pauses.
engine.gamepad.left=DpadLeft+LeftStickLeft
engine.gamepad.right=DpadRight+LeftStickRight
engine.gamepad.forward=DpadUp+LeftStickUp
engine.gamepad.back=DpadDown+LeftStickDown
engine.gamepad.jump=South
engine.gamepad.dive=East+West
engine.gamepad.pause=Start
